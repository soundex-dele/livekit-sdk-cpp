cmake_minimum_required(VERSION 3.24)
project(sdk_consumer LANGUAGES C CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(LIVEKIT_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(LIVEKIT_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LIVEKIT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LIVEKIT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LIVEKIT_ENABLE_VIDEO_PLAYOUT OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. sdk)
add_executable(consumer ../../examples/room_client.cpp)
target_link_libraries(consumer PRIVATE LiveKit::SDK)

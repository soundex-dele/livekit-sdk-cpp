cmake_minimum_required(VERSION 3.24)
project(sdk_consumer LANGUAGES C CXX)
set(SDK_CONSUMER_COMPONENT RTC CACHE STRING "RTC or ServerSDK")
if(SDK_CONSUMER_COMPONENT STREQUAL "ServerSDK")
  set(LIVEKIT_BUILD_RTC OFF CACHE BOOL "" FORCE)
  set(LIVEKIT_BUILD_PLAYER OFF CACHE BOOL "" FORCE)
  set(LIVEKIT_BUILD_SERVER_SDK ON CACHE BOOL "" FORCE)
elseif(SDK_CONSUMER_COMPONENT STREQUAL "RTC")
  set(LIVEKIT_BUILD_RTC ON CACHE BOOL "" FORCE)
else()
  message(FATAL_ERROR "Unknown SDK_CONSUMER_COMPONENT")
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(LIVEKIT_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(LIVEKIT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LIVEKIT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LIVEKIT_ENABLE_VIDEO_PLAYOUT OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. sdk)
if(SDK_CONSUMER_COMPONENT STREQUAL "ServerSDK")
  add_executable(consumer server_sdk.cpp)
  target_link_libraries(consumer PRIVATE LiveKit::ServerSDK)
  enable_testing()
  add_test(NAME server_sdk_consumer COMMAND consumer)
else()
  add_executable(consumer ../../examples/room_client/main.cpp)
  target_link_libraries(consumer PRIVATE LiveKit::SDK)
endif()

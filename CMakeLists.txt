cmake_minimum_required(VERSION 3.24)
project(livekit_sdk_cpp VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:/utf-8;/bigobj>")
endif()
option(LIVEKIT_BUILD_DEMO "Build ImGui conference viewer" ${PROJECT_IS_TOP_LEVEL})
option(LIVEKIT_BUILD_CLI "Build the SDK-based lk CLI" ${PROJECT_IS_TOP_LEVEL})
option(LIVEKIT_BUILD_EXAMPLES "Build public SDK examples" ${PROJECT_IS_TOP_LEVEL})
option(LIVEKIT_BUILD_TESTS "Build SDK and CLI tests" ${PROJECT_IS_TOP_LEVEL})
option(LIVEKIT_ENABLE_VIDEO_PLAYOUT "Enable optional SDL video windows" ON)
option(LIVEKIT_BUILD_RTC "Build the existing RTC SDK and its applications" ON)
option(LIVEKIT_BUILD_PLAYER "Build the independent local-file player" OFF)
option(LIVEKIT_BUILD_REMOTE_APP "Build adjacent ImGui remote-control host and test fixtures" OFF)
if(LIVEKIT_BUILD_PLAYER OR LIVEKIT_BUILD_RTC)
  include(cmake/MediaGraph.cmake)
endif()
if(LIVEKIT_BUILD_PLAYER)
  enable_testing()
  add_subdirectory(player)
endif()
if(NOT LIVEKIT_BUILD_RTC)
  return()
endif()
include(cmake/WebRTC.cmake)
include(cmake/Dependencies.cmake)
add_library(livekit_media_nodes_webrtc STATIC src/backends/webrtc/codec_factories.cpp
  src/backends/webrtc/graph_video_encoder.cpp src/backends/webrtc/graph_video_decoder.cpp)
add_library(LiveKit::MediaNodesWebRTC ALIAS livekit_media_nodes_webrtc)
target_include_directories(livekit_media_nodes_webrtc PRIVATE src)
target_link_libraries(livekit_media_nodes_webrtc PRIVATE LiveKit::MediaGraph graphflow::core WebRTC::webrtc)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
set(PROTO_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${PROTO_DIR}")
add_custom_command(OUTPUT "${PROTO_DIR}/livekit.pb.cc" "${PROTO_DIR}/livekit.pb.h"
  COMMAND protobuf::protoc --cpp_out=${PROTO_DIR} -I ${CMAKE_CURRENT_SOURCE_DIR}/protocol ${CMAKE_CURRENT_SOURCE_DIR}/protocol/livekit.proto
  DEPENDS protobuf::protoc protocol/livekit.proto VERBATIM)
add_library(livekit_protocol STATIC "${PROTO_DIR}/livekit.pb.cc")
target_include_directories(livekit_protocol PUBLIC "${PROTO_DIR}")
target_link_libraries(livekit_protocol PUBLIC protobuf::libprotobuf-lite)

add_library(livekit_sdk STATIC src/video_frames.cpp src/auth.cpp src/net.cpp src/process.cpp src/media.cpp src/media_nodes/capture.cpp src/media_nodes/recording.cpp src/room.cpp src/sdk.cpp src/room_service.cpp src/api.cpp src/services.cpp src/participant.cpp src/room_events.cpp)
if(WIN32)
  target_sources(livekit_sdk PRIVATE "${WEBRTC_SOURCE_DIR}/rtc_base/task_queue_win.cc")
  target_link_libraries(livekit_sdk PRIVATE gdi32 user32)
endif()
add_library(LiveKit::SDK ALIAS livekit_sdk)
target_sources(livekit_sdk PRIVATE src/media_nodes/capture_source.cpp)
target_sources(livekit_sdk PRIVATE src/remote_control.cpp src/control_input_win.cpp)
set_target_properties(livekit_sdk PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
target_compile_features(livekit_sdk PUBLIC cxx_std_20)
target_include_directories(livekit_sdk PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_include_directories(livekit_sdk PRIVATE src/internal src)
target_link_libraries(livekit_sdk PUBLIC nlohmann_json::nlohmann_json)
target_link_libraries(livekit_sdk PRIVATE WebRTC::webrtc livekit_protocol ixwebsocket LiveKit::MediaGraph livekit_media_nodes_webrtc graphflow::core)
if(LIVEKIT_ENABLE_VIDEO_PLAYOUT)
  target_sources(livekit_sdk PRIVATE src/video_playout.cpp)
  target_link_libraries(livekit_sdk PRIVATE SDL2::SDL2-static)
else()
  target_sources(livekit_sdk PRIVATE src/video_playout_disabled.cpp)
endif()
if(LIVEKIT_BUILD_CLI)
  add_executable(lk cli/main.cpp)
  set_target_properties(lk PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
  target_link_libraries(lk PRIVATE LiveKit::SDK CLI11::CLI11)
  if(WIN32)
    target_link_libraries(lk PRIVATE shell32)
  endif()
  install(TARGETS lk RUNTIME DESTINATION bin)
endif()
if(LIVEKIT_BUILD_EXAMPLES)
  add_executable(livekit_sdk_example examples/room_client.cpp)
  target_link_libraries(livekit_sdk_example PRIVATE LiveKit::SDK)
endif()
if(LIVEKIT_BUILD_TESTS)
  enable_testing()
  add_executable(livekit_remote_control_tests tests/remote_control_tests.cpp)
  target_include_directories(livekit_remote_control_tests PRIVATE src)
  target_link_libraries(livekit_remote_control_tests PRIVATE LiveKit::SDK)
  add_test(NAME remote_control COMMAND livekit_remote_control_tests)
  add_executable(livekit_media_graph_tests tests/media_graph_tests.cpp)
  target_include_directories(livekit_media_graph_tests PRIVATE src src/internal)
  target_link_libraries(livekit_media_graph_tests PRIVATE LiveKit::SDK livekit_media_nodes_webrtc LiveKit::MediaGraph graphflow::core WebRTC::webrtc)
  add_test(NAME media_graph COMMAND livekit_media_graph_tests)
  set_tests_properties(media_graph PROPERTIES TIMEOUT 45)
  add_executable(livekit_video_frames_tests tests/video_frames_tests.cpp)
  target_include_directories(livekit_video_frames_tests PRIVATE src/internal)
  target_link_libraries(livekit_video_frames_tests PRIVATE LiveKit::SDK WebRTC::webrtc)
  add_test(NAME video_frames COMMAND livekit_video_frames_tests)
  add_executable(lk_tests tests/core_tests.cpp)
  target_include_directories(lk_tests PRIVATE src/internal src)
  target_link_libraries(lk_tests PRIVATE LiveKit::SDK WebRTC::webrtc livekit_protocol graphflow::core)
  add_test(NAME core COMMAND lk_tests)
  add_executable(livekit_sdk_tests tests/sdk_tests.cpp)
  target_include_directories(livekit_sdk_tests PRIVATE src/internal)
  target_link_libraries(livekit_sdk_tests PRIVATE LiveKit::SDK)
  add_test(NAME sdk COMMAND livekit_sdk_tests)
  set_tests_properties(sdk PROPERTIES TIMEOUT 45)
  add_executable(livekit_sdk_integration tests/sdk_integration.cpp)
  target_link_libraries(livekit_sdk_integration PRIVATE LiveKit::SDK)
  if(LIVEKIT_BUILD_CLI)
    add_test(NAME cli_help COMMAND lk --help)
    add_test(NAME native_webrtc COMMAND lk doctor)
    set_tests_properties(native_webrtc PROPERTIES TIMEOUT 30)
  endif()
  if(LIVEKIT_ENABLE_VIDEO_PLAYOUT)
    add_executable(lk_video_tests tests/video_playout_tests.cpp)
    target_include_directories(lk_video_tests PRIVATE src/internal)
    target_link_libraries(lk_video_tests PRIVATE LiveKit::SDK WebRTC::webrtc SDL2::SDL2-static)
    add_test(NAME video_playout COMMAND lk_video_tests)
    set_tests_properties(video_playout PROPERTIES TIMEOUT 30 ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_RENDER_DRIVER=software")
  endif()
endif()

if(LIVEKIT_BUILD_DEMO)
  add_subdirectory(demo)
endif()
if(LIVEKIT_BUILD_REMOTE_APP)
  if(NOT WIN32 OR NOT TARGET SDL2::SDL2-static)
    message(FATAL_ERROR "Remote host requires Windows and LIVEKIT_ENABLE_VIDEO_PLAYOUT=ON")
  endif()
  add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../livekit-app/native" remote-app)
endif()

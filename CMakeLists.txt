cmake_minimum_required(VERSION 3.24)
project(livekit_sdk_cpp VERSION 0.3.0 LANGUAGES C CXX)
if(APPLE)
  # SDL and WebRTC both compile Objective-C sources. Enable their languages at
  # the common parent scope so sibling subdirectories share the compile rules.
  enable_language(OBJC OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 20)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
configure_file(cmake/version.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/public/livekit/version.hpp" @ONLY)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:/utf-8;/bigobj>")
endif()
option(LIVEKIT_BUILD_CLI "Build the SDK-based lk CLI" ${PROJECT_IS_TOP_LEVEL})
option(LIVEKIT_BUILD_EXAMPLES "Build public SDK examples" ${PROJECT_IS_TOP_LEVEL})
option(LIVEKIT_BUILD_TESTS "Build SDK and CLI tests" ${PROJECT_IS_TOP_LEVEL})
option(LIVEKIT_ENABLE_VIDEO_PLAYOUT "Enable optional SDL video windows" ON)
option(LIVEKIT_BUILD_RTC "Build the existing RTC SDK and its applications" ON)
option(LIVEKIT_RTC_FFMPEG_CODECS "Add software H.264/H.265 RTC codecs using an explicit FFmpeg installation" OFF)
if(APPLE AND LIVEKIT_BUILD_RTC)
  include(CheckCXXSourceCompiles)
  check_cxx_source_compiles("#include <thread>\n#include <stop_token>\nint main() { std::jthread t([](std::stop_token) {}); }" LIVEKIT_HAS_CXX20_THREADS)
  if(NOT LIVEKIT_HAS_CXX20_THREADS)
    message(FATAL_ERROR "RTC SDK requires C++20 std::jthread/stop_token. Use a macos preset with Homebrew llvm and its matching libc++.")
  endif()
endif()
option(LIVEKIT_BUILD_PLAYER "Build the independent local-file playback engine" OFF)
option(LIVEKIT_BUILD_STUDIO "Build Windows live production engine" OFF)
if(LIVEKIT_BUILD_STUDIO AND (NOT LIVEKIT_BUILD_RTC OR NOT LIVEKIT_BUILD_PLAYER))
  message(FATAL_ERROR "Studio requires RTC and Player engines")
endif()
option(LIVEKIT_BUILD_SERVER_SDK "Build trusted server management and token API" ${LIVEKIT_BUILD_RTC})
if(LIVEKIT_BUILD_PLAYER OR LIVEKIT_BUILD_RTC)
  include(cmake/MediaGraph.cmake)
endif()
if(LIVEKIT_BUILD_PLAYER)
  enable_testing()
  add_subdirectory(src/engines/player)
endif()
if(LIVEKIT_BUILD_RTC OR LIVEKIT_BUILD_SERVER_SDK)
  find_package(OpenSSL REQUIRED)
  include(cmake/Dependencies.cmake)
  add_library(livekit_server_sdk STATIC src/server_sdk/auth.cpp src/server_sdk/api.cpp
    src/server_sdk/services.cpp src/server_sdk/room_service.cpp)
  add_library(LiveKit::ServerSDK ALIAS livekit_server_sdk)
  target_compile_features(livekit_server_sdk PUBLIC cxx_std_20)
  target_include_directories(livekit_server_sdk PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" PRIVATE src)
  target_include_directories(livekit_server_sdk PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/public>")
  target_link_libraries(livekit_server_sdk PUBLIC nlohmann_json::nlohmann_json PRIVATE ixwebsocket OpenSSL::Crypto)
endif()
if(NOT LIVEKIT_BUILD_RTC)
  return()
endif()
include(cmake/WebRTC.cmake)
add_library(livekit_rtc_graph STATIC src/engines/rtc/rtc_graph.cpp)
target_include_directories(livekit_rtc_graph PRIVATE src)
target_link_libraries(livekit_rtc_graph PRIVATE LiveKit::MediaGraph graphflow::engine)
add_library(livekit_media_nodes_webrtc STATIC src/backends/webrtc/codec_factories.cpp
  src/backends/webrtc/graph_video_encoder.cpp src/backends/webrtc/graph_video_decoder.cpp)
add_library(LiveKit::MediaNodesWebRTC ALIAS livekit_media_nodes_webrtc)
target_include_directories(livekit_media_nodes_webrtc PRIVATE src)
target_link_libraries(livekit_media_nodes_webrtc PRIVATE LiveKit::MediaGraph livekit_rtc_graph graphflow::core WebRTC::webrtc)
if(LIVEKIT_RTC_FFMPEG_CODECS)
  include(cmake/PlayerFFmpeg.cmake)
  target_sources(livekit_media_nodes_webrtc PRIVATE src/backends/webrtc/ffmpeg_video_codecs.cpp)
  target_compile_definitions(livekit_media_nodes_webrtc PRIVATE LIVEKIT_RTC_FFMPEG_CODECS=1)
  target_link_libraries(livekit_media_nodes_webrtc PRIVATE livekit_ffmpeg_avcodec livekit_ffmpeg_avutil)
endif()
set(PROTO_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${PROTO_DIR}")
add_custom_command(OUTPUT "${PROTO_DIR}/livekit.pb.cc" "${PROTO_DIR}/livekit.pb.h"
  COMMAND protobuf::protoc --cpp_out=${PROTO_DIR} -I ${CMAKE_CURRENT_SOURCE_DIR}/protocol ${CMAKE_CURRENT_SOURCE_DIR}/protocol/livekit.proto
  DEPENDS protobuf::protoc protocol/livekit.proto VERBATIM)
add_library(livekit_protocol STATIC "${PROTO_DIR}/livekit.pb.cc")
target_include_directories(livekit_protocol PUBLIC "${PROTO_DIR}")
target_link_libraries(livekit_protocol PUBLIC protobuf::libprotobuf-lite)

add_library(livekit_sdk STATIC src/media_nodes/render/video_frames.cpp src/support/net.cpp src/support/process.cpp src/engines/rtc/media.cpp src/media_nodes/capture/capture.cpp src/media_nodes/recording/recording.cpp src/engines/rtc/room.cpp src/engines/rtc/sdk.cpp src/engines/rtc/participant.cpp src/engines/rtc/room_events.cpp)
if(WIN32)
  target_sources(livekit_sdk PRIVATE "${WEBRTC_SOURCE_DIR}/rtc_base/task_queue_win.cc")
  target_link_libraries(livekit_sdk PRIVATE gdi32 user32)
endif()
add_library(LiveKit::SDK ALIAS livekit_sdk)

if(LIVEKIT_BUILD_STUDIO)
  add_subdirectory(src/engines/studio)
endif()
target_sources(livekit_sdk PRIVATE src/engines/rtc/signal_session.cpp)
target_sources(livekit_sdk PRIVATE src/engines/rtc/rpc_session.cpp)
target_sources(livekit_sdk PRIVATE src/media_nodes/render/video_stream.cpp)
target_sources(livekit_sdk PRIVATE src/media_nodes/render/audio_stream.cpp)
target_sources(livekit_sdk PRIVATE src/media_nodes/capture/capture_source.cpp)

target_sources(livekit_sdk PRIVATE src/media_nodes/capture/audio_source.cpp)

target_sources(livekit_sdk PRIVATE src/scenarios/remote_control/remote_control.cpp src/backends/platform/windows/control_input_win.cpp)
set_target_properties(livekit_sdk PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
target_compile_features(livekit_sdk PUBLIC cxx_std_20)
target_include_directories(livekit_sdk PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
target_include_directories(livekit_sdk PRIVATE src)
target_link_libraries(livekit_sdk PUBLIC nlohmann_json::nlohmann_json LiveKit::ServerSDK)
target_link_libraries(livekit_sdk PRIVATE WebRTC::webrtc livekit_protocol ixwebsocket LiveKit::MediaGraph livekit_rtc_graph livekit_media_nodes_webrtc graphflow::core)
if(LIVEKIT_ENABLE_VIDEO_PLAYOUT)
  target_sources(livekit_sdk PRIVATE src/media_nodes/render/video_playout.cpp)
  target_link_libraries(livekit_sdk PRIVATE SDL2::SDL2-static)
else()
  target_sources(livekit_sdk PRIVATE src/media_nodes/render/video_playout_disabled.cpp)
endif()

  add_executable(lk tools/cli/main.cpp)
  set_target_properties(lk PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
  target_link_libraries(lk PRIVATE LiveKit::SDK CLI11::CLI11)
  if(WIN32)
    target_link_libraries(lk PRIVATE shell32)
  endif()

  target_link_libraries(livekit_media_graph_tests PRIVATE LiveKit::SDK livekit_media_nodes_webrtc livekit_rtc_graph LiveKit::MediaGraph graphflow::core WebRTC::webrtc)

  add_executable(livekit_sdk_integration tests/sdk_integration.cpp)
  target_link_libraries(livekit_sdk_integration PRIVATE LiveKit::SDK)
  add_executable(livekit_uav_requirements tests/uav_requirements.cpp)

  if(LIVEKIT_RTC_FFMPEG_CODECS)
    livekit_player_runtime(livekit_uav_requirements)

  add_executable(livekit_reconnect_integration tests/reconnect_integration.cpp)
  target_link_libraries(livekit_reconnect_integration PRIVATE LiveKit::SDK)

    add_test(NAME cli_help COMMAND lk --help)
    add_test(NAME native_webrtc COMMAND lk doctor)
    set_tests_properties(native_webrtc PROPERTIES TIMEOUT 30)
  endif()
  if(LIVEKIT_ENABLE_VIDEO_PLAYOUT)
    add_executable(lk_video_tests tests/video_playout_tests.cpp)
    target_include_directories(lk_video_tests PRIVATE src)
    target_link_libraries(lk_video_tests PRIVATE LiveKit::SDK WebRTC::webrtc SDL2::SDL2-static)

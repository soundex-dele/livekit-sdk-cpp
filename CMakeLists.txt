# Optional matching host protoc avoids rebuilding the compiler for each product.
set(LIVEKIT_PROTOC_EXECUTABLE "" CACHE FILEPATH "Existing protoc 3.21.12 executable")
set(_sdk_protoc protobuf::protoc)
if(LIVEKIT_PROTOC_EXECUTABLE)
  execute_process(COMMAND "${LIVEKIT_PROTOC_EXECUTABLE}" --version OUTPUT_VARIABLE _protoc_version RESULT_VARIABLE _protoc_result OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT _protoc_result EQUAL 0 OR NOT _protoc_version STREQUAL "libprotoc 3.21.12")
    message(FATAL_ERROR "LIVEKIT_PROTOC_EXECUTABLE must be protoc 3.21.12")
  endif()
  set(_sdk_protoc "${LIVEKIT_PROTOC_EXECUTABLE}")
endif()
add_custom_command(OUTPUT "${PROTO_DIR}/livekit.pb.cc" "${PROTO_DIR}/livekit.pb.h"
  COMMAND ${_sdk_protoc} --cpp_out=${PROTO_DIR} -I ${CMAKE_CURRENT_SOURCE_DIR}/protocol ${CMAKE_CURRENT_SOURCE_DIR}/protocol/livekit.proto
  DEPENDS ${_sdk_protoc} protocol/livekit.proto VERBATIM)
add_library(livekit_protocol STATIC "${PROTO_DIR}/livekit.pb.cc")

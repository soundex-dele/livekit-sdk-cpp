if(NOT WIN32 AND NOT (APPLE AND CMAKE_SYSTEM_NAME STREQUAL "Darwin"))
  message(FATAL_ERROR "Studio requires Windows or macOS")
endif()
if(APPLE AND CMAKE_OSX_DEPLOYMENT_TARGET AND CMAKE_OSX_DEPLOYMENT_TARGET VERSION_LESS "14.0")
  message(FATAL_ERROR "Studio requires macOS 14 or later")
endif()
if(NOT TARGET LiveKit::MediaInputs OR NOT TARGET LiveKit::Player)
  message(FATAL_ERROR "Studio requires media inputs and LIVEKIT_BUILD_PLAYER")
endif()
add_library(livekit_studio STATIC studio.cpp studio_graph.cpp compositor.cpp recorder.cpp)
add_library(LiveKit::Studio ALIAS livekit_studio)
livekit_without_exceptions(livekit_studio)
target_include_directories(livekit_studio PRIVATE "${PROJECT_SOURCE_DIR}/src" "${LIVEKIT_FFMPEG_INCLUDE}")
target_link_libraries(livekit_studio PUBLIC LiveKit::MediaInputs nlohmann_json::nlohmann_json PRIVATE LiveKit::Player graphflow::engine WebRTC::webrtc livekit_ffmpeg_avformat livekit_ffmpeg_avcodec livekit_ffmpeg_avutil livekit_ffmpeg_swscale livekit_ffmpeg_swresample)
if(WIN32)
  # C++/WinRT HRESULT exceptions are translated only at this adapter boundary.
  add_library(livekit_studio_winrt OBJECT capture_windows.cpp)
  target_include_directories(livekit_studio_winrt PRIVATE "${PROJECT_SOURCE_DIR}/src" "${LIVEKIT_FFMPEG_INCLUDE}")
  target_link_libraries(livekit_studio_winrt PRIVATE LiveKit::MediaInputs WebRTC::webrtc)
  target_compile_definitions(livekit_studio_winrt PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(livekit_studio_winrt PRIVATE /EHsc)
  endif()
  target_sources(livekit_studio PRIVATE $<TARGET_OBJECTS:livekit_studio_winrt> audio_windows.cpp)
  target_link_libraries(livekit_studio PRIVATE d3d11 dxgi d3dcompiler windowsapp windowscodecs ole32 uuid gdi32 mmdevapi)
  target_compile_definitions(livekit_studio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
elseif(APPLE)
  target_sources(livekit_studio PRIVATE compositor_macos.mm capture_macos.mm)
  target_compile_options(livekit_studio PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(livekit_studio PRIVATE "-framework Metal" "-framework AppKit" "-framework ScreenCaptureKit" "-framework AVFoundation" "-framework AudioToolbox" "-framework CoreMedia" "-framework CoreVideo" "-framework ImageIO")
endif()

if(LIVEKIT_BUILD_TESTS)
  add_executable(livekit_studio_result_tests "${PROJECT_SOURCE_DIR}/tests/studio_result_tests.cpp")
  target_include_directories(livekit_studio_result_tests PRIVATE "${PROJECT_SOURCE_DIR}/src")
  target_link_libraries(livekit_studio_result_tests PRIVATE LiveKit::Studio graphflow::engine)
  if(WIN32)
    target_link_libraries(livekit_studio_result_tests PRIVATE ole32)
  endif()
  livekit_without_exceptions(livekit_studio_result_tests)
  livekit_player_runtime(livekit_studio_result_tests)
  add_test(NAME studio_result_contracts COMMAND livekit_studio_result_tests)
  set_tests_properties(studio_result_contracts PROPERTIES TIMEOUT 45)
endif()

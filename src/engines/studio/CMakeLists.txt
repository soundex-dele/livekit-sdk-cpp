if(NOT WIN32 AND NOT (APPLE AND CMAKE_SYSTEM_NAME STREQUAL "Darwin"))
  message(FATAL_ERROR "Studio requires Windows or macOS")
endif()
if(APPLE AND CMAKE_OSX_DEPLOYMENT_TARGET AND CMAKE_OSX_DEPLOYMENT_TARGET VERSION_LESS "14.0")
  message(FATAL_ERROR "Studio requires macOS 14 or later")
endif()
if(NOT TARGET LiveKit::SDK OR NOT TARGET LiveKit::Player)
  message(FATAL_ERROR "Studio requires LIVEKIT_BUILD_RTC and LIVEKIT_BUILD_PLAYER")
endif()
add_library(livekit_studio STATIC studio.cpp studio_graph.cpp compositor.cpp recorder.cpp)
add_library(LiveKit::Studio ALIAS livekit_studio)
target_include_directories(livekit_studio PRIVATE "${PROJECT_SOURCE_DIR}/src" "${LIVEKIT_FFMPEG_INCLUDE}")
target_link_libraries(livekit_studio PUBLIC LiveKit::SDK PRIVATE LiveKit::Player graphflow::engine WebRTC::webrtc livekit_ffmpeg_avformat livekit_ffmpeg_avcodec livekit_ffmpeg_avutil livekit_ffmpeg_swscale livekit_ffmpeg_swresample)
if(WIN32)
  target_sources(livekit_studio PRIVATE capture_windows.cpp audio_windows.cpp)
  target_link_libraries(livekit_studio PRIVATE d3d11 dxgi d3dcompiler windowsapp windowscodecs ole32 uuid gdi32 mmdevapi)
  target_compile_definitions(livekit_studio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
elseif(APPLE)
  target_sources(livekit_studio PRIVATE compositor_macos.mm capture_macos.mm)
  target_compile_options(livekit_studio PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_libraries(livekit_studio PRIVATE "-framework Metal" "-framework AppKit" "-framework ScreenCaptureKit" "-framework AVFoundation" "-framework AudioToolbox" "-framework CoreMedia" "-framework CoreVideo" "-framework ImageIO")
endif()
